# Core library: public headers (include/sbmc/) and sbmc_core (rudy reader, Ising model in int64,
# dense INT8 rows of J, Philox streams, initial positions, parts of simulated bifurcation common
# to every variant: pressure schedule and automatic c0; parts of simulated annealing common to
# every variant: schedule of beta and acceptance threshold; results schema 1, the replica file,
# CSV helpers and the table of best-known values). Standard C++20 and OpenMP only: the public
# headers contain no CUDA type and include no CUDA header; the only CUDA-related element is the
# macro SBMC_HD of sbmc/philox.hpp, active under __CUDACC__ only (sb_sign and sb_update of
# sbmc/sb.hpp, sa_initial_spin and sa_acceptance_word of sbmc/sa.hpp use it).

add_library(sbmc_headers INTERFACE)
add_library(sbmc::headers ALIAS sbmc_headers)
target_include_directories(sbmc_headers INTERFACE "${PROJECT_SOURCE_DIR}/include")

add_library(sbmc_core STATIC instance.cpp rudy.cpp initial_state.cpp sb.cpp sa.cpp results.cpp)
add_library(sbmc::core ALIAS sbmc_core)
target_link_libraries(sbmc_core PUBLIC sbmc::headers PRIVATE sbmc::flags OpenMP::OpenMP_CXX)

# Header check. For every public header (explicit list, no file(GLOB)), CMake generates in the
# build tree a translation unit that includes the header twice and does nothing else. The units
# are compiled as C++ in the OBJECT library sbmc_header_check, part of the default build of every
# preset: a CUDA type or header in include/sbmc/ breaks the build of the cpu presets. With
# SBMC_ENABLE_CUDA the same units are also compiled as .cu files, with the architectures of the
# preset, in sbmc_header_check_cuda: a header that nvcc cannot compile breaks the build of the
# cuda and t4 presets. file(CONFIGURE) rewrites a unit only when its content changes.
set(_sbmc_public_headers
  error.hpp
  philox.hpp
  instance.hpp
  rudy.hpp
  initial_state.hpp
  sb.hpp
  sb_sequential.hpp
  sb_openmp.hpp
  sa.hpp
  sa_sequential.hpp
  sa_openmp.hpp
  bruteforce.hpp
  results.hpp
)
set(_sbmc_header_check_dir "${CMAKE_CURRENT_BINARY_DIR}/header_check")
set(_sbmc_header_check_cxx_sources "")
set(_sbmc_header_check_cuda_sources "")
foreach(_sbmc_header IN LISTS _sbmc_public_headers)
  string(REPLACE "." "_" _sbmc_stem "${_sbmc_header}")
  string(CONCAT _sbmc_unit
    "// Generated by src/core/CMakeLists.txt: header check of <sbmc/${_sbmc_header}>.\n"
    "#include <sbmc/${_sbmc_header}>\n"
    "#include <sbmc/${_sbmc_header}>\n")
  file(CONFIGURE OUTPUT "${_sbmc_header_check_dir}/${_sbmc_stem}.cpp"
       CONTENT "${_sbmc_unit}" @ONLY)
  list(APPEND _sbmc_header_check_cxx_sources "${_sbmc_header_check_dir}/${_sbmc_stem}.cpp")
  if(SBMC_ENABLE_CUDA)
    file(CONFIGURE OUTPUT "${_sbmc_header_check_dir}/${_sbmc_stem}.cu"
         CONTENT "${_sbmc_unit}" @ONLY)
    list(APPEND _sbmc_header_check_cuda_sources "${_sbmc_header_check_dir}/${_sbmc_stem}.cu")
  endif()
endforeach()

add_library(sbmc_header_check OBJECT ${_sbmc_header_check_cxx_sources})
target_link_libraries(sbmc_header_check PRIVATE sbmc::flags sbmc::headers)

if(SBMC_ENABLE_CUDA)
  add_library(sbmc_header_check_cuda OBJECT ${_sbmc_header_check_cuda_sources})
  target_link_libraries(sbmc_header_check_cuda PRIVATE sbmc::flags sbmc::headers CUDA::cudart)
endif()
